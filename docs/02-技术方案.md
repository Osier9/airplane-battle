### 2.1 Game.ts - 主游戏类

```typescript
class Game {
  private scene: Scene3D;
  private gameState: GameState;
  private inputSystem: InputSystem;
  private collisionSystem: CollisionSystem;
  private audioSystem: AudioSystem;
  private levelSystem: LevelSystem;
  private uiManager: UIManager;
  private animationFrameId: number;

  constructor(canvasElement: HTMLCanvasElement) {
    this.scene = new Scene3D(canvasElement);
    this.gameState = new GameState();
    this.inputSystem = new InputSystem();
    this.collisionSystem = new CollisionSystem();
    this.audioSystem = new AudioSystem();
    this.levelSystem = new LevelSystem();
    this.uiManager = new UIManager();
  }

  init(): void {
    // 初始化游戏
    this.scene.init();
    this.levelSystem.loadLevel(1);
    this.audioSystem.playBGM();
    this.startGameLoop();
  }

  private startGameLoop(): void {
    const gameLoop = () => {
      this.update();
      this.render();
      this.animationFrameId = requestAnimationFrame(gameLoop);
    };
    gameLoop();
  }

  private update(deltaTime: number): void {
    // 1. 处理输入
    const input = this.inputSystem.getInput();
    
    // 2. 更新玩家
    this.gameState.player.update(input, deltaTime);
    
    // 3. 更新敌人
    this.gameState.enemies.forEach(enemy => {
      enemy.update(this.gameState.player, deltaTime);
    });
    
    // 4. 碰撞检测
    this.collisionSystem.check(this.gameState);
    
    // 5. 更新游戏状态
    this.gameState.update(deltaTime);
    
    // 6. 更新UI
    this.uiManager.update(this.gameState);
    
    // 7. 检查关卡完成
    this.checkLevelCompletion();
  }

  private render(): void {
    this.scene.render();
  }

  private checkLevelCompletion(): void {
    if (this.gameState.checkLevelComplete()) {
      this.onLevelComplete();
    }
    if (this.gameState.checkGameOver()) {
      this.onGameOver();
    }
  }

  pause(): void {
    this.gameState.isPaused = true;
  }

  resume(): void {
    this.gameState.isPaused = false;
  }

  destroy(): void {
    cancelAnimationFrame(this.animationFrameId);
    this.scene.dispose();
    this.audioSystem.dispose();
  }
}
```

### 2.2 Scene3D.ts - 3D场景管理

```typescript
class Scene3D {
  private scene: THREE.Scene;
  private camera: THREE.PerspectiveCamera;
  private renderer: THREE.WebGLRenderer;
  private canvas: HTMLCanvasElement;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.init3D();
  }

  private init3D(): void {
    // 创建场景
    this.scene = new THREE.Scene();
    this.scene.background = new THREE.Color(0x0a0e27);
    this.scene.fog = new THREE.Fog(0x0a0e27, 100, 200);

    // 创建相机
    this.camera = new THREE.PerspectiveCamera(
      75,
      this.canvas.clientWidth / this.canvas.clientHeight,
      0.1,
      1000
    );
    this.camera.position.set(0, 10, 15);
    this.camera.lookAt(0, 0, 0);

    // 创建渲染器
    this.renderer = new THREE.WebGLRenderer({ 
      canvas: this.canvas,
      antialias: true,
      alpha: true 
    });
    this.renderer.setSize(this.canvas.clientWidth, this.canvas.clientHeight);
    this.renderer.setPixelRatio(window.devicePixelRatio);
    this.renderer.shadowMap.enabled = true;

    // 添加光源
    this.setupLights();

    // 创建地面
    this.createGround();

    // 响应窗口大小变化
    window.addEventListener('resize', () => this.onWindowResize());
  }

  private setupLights(): void {
    // 环境光
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
    this.scene.add(ambientLight);

    // 平行光（主光源）
    const directionalLight = new THREE.DirectionalLight(0x64b5ff, 0.8);
    directionalLight.position.set(50, 50, 50);
    directionalLight.shadow.mapSize.width = 2048;
    directionalLight.shadow.mapSize.height = 2048;
    directionalLight.castShadow = true;
    this.scene.add(directionalLight);

    // 点光源（氛围灯）
    const pointLight = new THREE.PointLight(0xff00ff, 0.5, 100);
    pointLight.position.set(30, 30, 30);
    this.scene.add(pointLight);
  }

  private createGround(): void {
    // 创建网格地面（科幻风格）
    const geometry = new THREE.PlaneGeometry(100, 100, 20, 20);
    const material = new THREE.LineBasicMaterial({
      color: 0x00ff88,
      wireframe: true
    });
    const grid = new THREE.LineSegments(geometry, material);
    grid.rotation.x = -Math.PI / 2;
    this.scene.add(grid);

    // 添加地面平面（碰撞用）
    const groundGeo = new THREE.PlaneGeometry(100, 100);
    const groundMat = new THREE.MeshStandardMaterial({
      color: 0x1a1a2e,
      transparent: true,
      opacity: 0.3
    });
    const ground = new THREE.Mesh(groundGeo, groundMat);
    ground.rotation.x = -Math.PI / 2;
    ground.receiveShadow = true;
    this.scene.add(ground);
  }

  addObject(object: THREE.Object3D): void {
    this.scene.add(object);
  }

  removeObject(object: THREE.Object3D): void {
    this.scene.remove(object);
  }

  updateCamera(playerPosition: Vector3): void {
    const cameraDistance = 15;
    const cameraHeight = 10;
    
    this.camera.position.lerp(
      new THREE.Vector3(playerPosition.x, cameraHeight, playerPosition.z + cameraDistance),
      0.1
    );
    this.camera.lookAt(playerPosition.x, 1, playerPosition.z);
  }

  render(): void {
    this.renderer.render(this.scene, this.camera);
  }

  dispose(): void {
    this.renderer.dispose();
  }

  private onWindowResize(): void {
    const width = this.canvas.clientWidth;
    const height = this.canvas.clientHeight;
    this.camera.aspect = width / height;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(width, height);
  }
}
```

### 2.3 Player.ts - 玩家飞机

```typescript
class Player {
  public position: Vector3;
  public velocity: Vector3;
  public health: number;
  public shield: number;
  public bombs: number;
  private model: THREE.Object3D;
  private moveSpeed: number = 0.25;
  private shootCooldown: number = 0;
  private specialShootCooldown: number = 0;

  constructor(position: Vector3) {
    this.position = position.clone();
    this.velocity = new Vector3(0, 0, 0);
    this.health = 100;
    this.shield = 0;
    this.bombs = 3;
    this.createModel();
  }

  private createModel(): void {
    const geometry = new THREE.ConeGeometry(1, 2, 8);
    const material = new THREE.MeshStandardMaterial({
      color: 0x00ff88,
      emissive: 0x00ff88,
      emissiveIntensity: 0.5
    });
    this.model = new THREE.Mesh(geometry, material);
    this.model.castShadow = true;
    this.model.receiveShadow = true;
  }

  update(input: InputState, deltaTime: number): void {
    // 处理移动输入
    const moveVector = new Vector3(0, 0, 0);
    
    if (input.moveUp) moveVector.z -= 1;
    if (input.moveDown) moveVector.z += 1;
    if (input.moveLeft) moveVector.x -= 1;
    if (input.moveRight) moveVector.x += 1;

    // 规范化并应用速度
    if (moveVector.length() > 0) {
      moveVector.normalize();
      this.velocity = moveVector.multiplyScalar(this.moveSpeed);
    }

    // 应用边界限制
    this.position.x = Math.max(-40, Math.min(40, this.position.x + this.velocity.x));
    this.position.z = Math.max(-40, Math.min(40, this.position.z + this.velocity.z));

    // 更新模型位置
    this.model.position.copy(this.position);

    // 更新射击冷却
    this.shootCooldown -= deltaTime;
    this.specialShootCooldown -= deltaTime;
  }

  shoot(): Bullet | null {
    if (this.shootCooldown <= 0) {
      this.shootCooldown = 0.2; // 200ms冷却
      const direction = new Vector3(0, 0, -1);
      return new Bullet(this.position.clone(), direction, 10); // 伤害10
    }
    return null;
  }

  specialShoot(): Bullet[] | null {
    if (this.specialShootCooldown <= 0) {
      this.specialShootCooldown = 1.0; // 1秒冷却
      const bullets: Bullet[] = [];
      for (let i = 0; i < 3; i++) {
        const offset = new Vector3((i - 1) * 2, 0, 0);
        const direction = new Vector3(0, 0, -1);
        bullets.push(new Bullet(this.position.clone().add(offset), direction, 50));
      }
      return bullets;
    }
    return null;
  }

  useBomb(): boolean {
    if (this.bombs > 0) {
      this.bombs--;
      return true;
    }
    return false;
  }

  takeDamage(damage: number): void {
    if (this.shield > 0) {
      this.shield -= damage;
      if (this.shield < 0) {
        this.health += this.shield;
        this.shield = 0;
      }
    } else {
      this.health -= damage;
    }
  }

  addHealth(amount: number): void {
    this.health = Math.min(100, this.health + amount);
  }

  addShield(amount: number): void {
    this.shield = Math.min(100, this.shield + amount);
  }

  addBomb(amount: number): void {
    this.bombs = Math.min(5, this.bombs + amount);
  }

  getModel(): THREE.Object3D {
    return this.model;
  }

  isAlive(): boolean {
    return this.health > 0;
  }
}
```

### 3.1 碰撞检测系统

```typescript
class CollisionSystem {
  check(gameState: GameState): void {
    const player = gameState.player;

    // 玩家与敌人子弹的碰撞
    gameState.enemyBullets.forEach((bullet, index) => {
      if (this.checkSphereSphere(player.position, 1, bullet.position, 0.3)) {
        player.takeDamage(bullet.damage);
        gameState.enemyBullets.splice(index, 1);
      }
    });

    // 玩家与敌人的碰撞
    gameState.enemies.forEach((enemy) => {
      if (this.checkSphereSphere(player.position, 1, enemy.position, 1)) {
        player.takeDamage(5);
        enemy.takeDamage(10);
      }
    });

    // 玩家子弹与敌人的碰撞
    gameState.playerBullets.forEach((bullet, bulletIndex) => {
      gameState.enemies.forEach((enemy, enemyIndex) => {
        if (this.checkSphereSphere(bullet.position, 0.3, enemy.position, 1)) {
          enemy.takeDamage(bullet.damage);
          gameState.playerBullets.splice(bulletIndex, 1);
          
          if (!enemy.isAlive()) {
            gameState.score += enemy.getReward();
            gameState.killedEnemies++;
          }
        }
      });
    });

    // 玩家与道具的碰撞
    gameState.pickups.forEach((pickup, index) => {
      if (this.checkSphereSphere(player.position, 1, pickup.position, 0.5)) {
        pickup.apply(player);
        gameState.pickups.splice(index, 1);
      }
    });
  }

  private checkSphereSphere(
    pos1: Vector3,
    radius1: number,
    pos2: Vector3,
    radius2: number
  ): boolean {
    const distance = pos1.distance(pos2);
    return distance < (radius1 + radius2);
  }
}
```

### 3.2 粒子系统

```typescript
class ParticleSystem {
  private particles: Particle[] = [];
  private scene: THREE.Scene;

  constructor(scene: THREE.Scene) {
    this.scene = scene;
  }

  createExplosion(position: Vector3, color: number = 0xff6b00): void {
    const particleCount = 30;
    for (let i = 0; i < particleCount; i++) {
      const angle = (Math.PI * 2 * i) / particleCount;
      const velocity = new Vector3(
        Math.cos(angle) * 0.5,
        Math.random() * 0.5,
        Math.sin(angle) * 0.5
      );

      const particle = new Particle(
        position.clone(),
        velocity,
        color,
        0.5 // 存活时间
      );
      this.particles.push(particle);
      this.scene.add(particle.getModel());
    }
  }

  createTrail(position: Vector3, direction: Vector3): void {
    const particle = new Particle(
      position.clone(),
      direction.multiplyScalar(0.1),
      0x00ff88,
      0.2
    );
    this.particles.push(particle);
    this.scene.add(particle.getModel());
  }

  update(deltaTime: number): void {
    for (let i = this.particles.length - 1; i >= 0; i--) {
      const particle = this.particles[i];
      particle.update(deltaTime);

      if (particle.isDead()) {
        this.scene.remove(particle.getModel());
        this.particles.splice(i, 1);
      }
    }
  }
}
```

```typescript
    }));
  }

    this.bgm.play();
  }

  stopBGM(): void {
    this.bgm.stop();
  }

    const sound = this.sfx.get(name);
    if (sound) {
      sound.play();
    }
  }

    Howler.volume(volume);
  }

  dispose(): void {
    this.bgm.unload();
    this.sfx.forEach(sound => sound.unload());
  }
}
```

## 4. 关卡系统实现

```typescript
class LevelSystem {
  private levels: LevelConfig[];

  constructor() {
    this.initializeLevels();
  }

  private initializeLevels(): void {
    this.levels = [
      // 普通关卡 1-10
      { level: 1, targetKills: 10, timeLimit: 120, enemies: [1, 1, 1] },
      { level: 2, targetKills: 10, timeLimit: 120, enemies: [1, 1, 1, 1] },
      { level: 3, targetKills: 20, timeLimit: 120, enemies: [1, 2, 1] },
      { level: 4, targetKills: 20, timeLimit: 120, enemies: [1, 2, 2] },
      { level: 5, targetKills: 30, timeLimit: 150, enemies: [1, 2, 3] },
      { level: 6, targetKills: 30, timeLimit: 150, enemies: [2, 2, 3] },
      { level: 7, targetKills: 40, timeLimit: 150, enemies: [2, 3, 3] },
      { level: 8, targetKills: 40, timeLimit: 150, enemies: [3, 3, 3] },
      { level: 9, targetKills: 50, timeLimit: 180, enemies: [3, 3, 3, 3] },
      { level: 10, targetKills: 50, timeLimit: 180, enemies: [3, 3, 3, 3, 3] },
      // Boss关卡 11-20
      { level: 11, isBoss: true, bossId: 1, timeLimit: 180 },
      { level: 12, isBoss: true, bossId: 2, timeLimit: 180 },
      // ... 继续到20
    ];
  }

  getLevel(levelNumber: number): LevelConfig {
    return this.levels[levelNumber - 1];
  }

  generateEnemies(levelNumber: number): Enemy[] {
    const config = this.getLevel(levelNumber);
    const enemies: Enemy[] = [];

    if (config.isBoss) {
      enemies.push(new Boss(config.bossId));
    } else {
      config.enemies?.forEach(type => {
        const enemy = this.createEnemyByType(type);
        enemies.push(enemy);
      });
    }

    return enemies;
  }

  private createEnemyByType(type: number): Enemy {
    switch (type) {
      case 1:
        return new ScoutEnemy(new Vector3(0, 2, -30));
      case 2:
        return new FighterEnemy(new Vector3(0, 2, -30));
      case 3:
        return new BomberEnemy(new Vector3(0, 2, -30));
      default:
        return new ScoutEnemy(new Vector3(0, 2, -30));
    }
  }
}
```

## 5. 关键实现技巧

### 5.1 3D模型优化
- 使用低多边形模型确保性能
- 通过材质和光效提升视觉效果
- 使用LOD（细节级别）系统远处敌人

### 5.2 粒子效果优化
- 使用对象池模式管理粒子
- 限制粒子数量在200-500之间
- 使用纹理贴图而非几何体渲染

### 5.3 性能优化
- 启用`requestAnimationFrame`优化帧率
- 使用视锥体剔除（frustum culling）
- 动态调整LOD和粒子质量

### 5.4 音频优化
- 预加载音效
- 使用Web Audio API的过滤和混音
- 限制同时播放音效数量

## 8. 常见问题解决

### Q: 如何处理不同屏幕尺寸？
A: 使用响应式设计，监听窗口resize事件，更新相机宽高比和渲染器大小。

### Q: 如何优化敌人AI？
A: 使用简单的状态机（追踪、射击、躲避），避免每帧复杂计算。

### Q: 如何保证跨浏览器兼容性？
A: 使用webgl2标准，polyfill必要的API，测试主流浏览器。
